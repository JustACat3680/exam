cmake_minimum_required(VERSION 3.31)
project(sum)
enable_testing()
find_package(GTest REQUIRED)
add_executable(sum main.cpp)
target_include_directories(sum PUBLIC include)
add_executable(test1 test.cpp)
target_include_directories(test1 PUBLIC include)
target_link_libraries(test1 PRIVATE GTest::gtest GTest::gmock GTest::gmock_main)
include(GoogleTest)
gtest_discover_tests(test1)
